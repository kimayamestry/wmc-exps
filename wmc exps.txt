def digital_ber(num_bits, snr_db, scheme): # to calculate bit rate error
    """BER for on-off ASK (OOK) or BPSK over AWGN at a given average SNR."""
    bits = rng.integers(0, 2, num_bits) # genrate a sequnce of random bits ie a sequn

    if scheme == "PSK":                  # BPSK: 0 -> -1, 1 -> +1 (avg energy = 1)
        symbols = 2 * bits - 1.0  # converts 1 to +1 and 0 to -1
        avg_energy = 1.0
        threshold = 0.0 # means if recived signal > 0 means 1 else 0
    elif scheme == "ASK":                # OOK: 0 -> 0, 1 -> +sqrt(2) (avg energy = 1)
        amp = np.sqrt(2.0)
        symbols = bits * amp
        avg_energy = 1.0
        threshold = amp / 2 # means if recived signal > amp/2 means 1 else 0
    else:
        raise ValueError(scheme)

    snr_linear = 10 ** (snr_db / 10)
    noise_std = np.sqrt(avg_energy / (2 * snr_linear)) # to calculate noise stnadard
    received = symbols + rng.normal(0, noise_std, num_bits) # add noise to the reciev
    detected = (received > threshold).astype(int) # will covert the signal to 0 and 1
    return np.mean(detected != bits) # to mark the error bits and basically count ther

def bpsk_ber(num_bits, snr_db): # calculates ber of bpsk
    bits = rng.integers(0, 2, num_bits) # generate random bits
    symbols = 2 * bits - 1.0  # BPSK: 0 -> -1, 1 -> +1 (avg energy = 1) convert b
    snr_linear = 10 ** (snr_db / 10)
    noise = rng.normal(0, np.sqrt(1 / (2 * snr_linear)), num_bits) # genearte noi
    detected = (symbols + noise > 0).astype(int) # add noise to the message and g
    return np.mean(detected != bits) # compare with original signal and mark the

def qpsk_ber(num_bits, snr_db):
    """
    Gray-coded QPSK: two bits per symbol on I and Q.
    SNR here is per-SYMBOL (Es/N0), same total symbol energy as BPSK,
    so each bit only gets half the energy -> worse BER at the same SNR.
    """
    num_bits -= num_bits % 2 # make number of bits even as we are using qpsk
    bits = rng.integers(0, 2, num_bits) # generate bits
    i_bits, q_bits = bits[0::2], bits[1::2] # divide the bits into to components
    # unit symbol energy split across I and Q
    i_sym = (2 * i_bits - 1) / np.sqrt(2) # +1/√2 for +1 , -1/√2 for -1 so that a
    q_sym = (2 * q_bits - 1) / np.sqrt(2)

    snr_linear = 10 ** (snr_db / 10)
    noise_std = np.sqrt(1 / (2 * snr_linear)) # create the noise
    i_rx = i_sym + rng.normal(0, noise_std, len(i_sym)) # add noise to simulate r
    q_rx = q_sym + rng.normal(0, noise_std, len(q_sym))
    det = np.empty(num_bits, dtype=int)
    det[0::2] = (i_rx > 0).astype(int) # detect the messeage by convert
    det[1::2] = (q_rx > 0).astype(int)
    return np.mean(det != bits) # return ber

num_bits_big = 1_000_000
snr_range_db = np.arange(0, 13, 1)
snr_linear = 10 ** (snr_range_db / 10)

ber_bpsk = [bpsk_ber(num_bits_big, s) for s in snr_range_db]

Exp3: To visualize and understand how cellular networks reuse limited spectrum across multiple geographic cells and compare N=3 and N=7 based on:


# %%
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.image as mpimg
from matplotlib.patches import RegularPolygon

LETTERS = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"

# %%
# ----------------------------------------------------------------------
# STEP 1 - Hexagonal geometry: coverts axial (q, r) -> Cartesian (x, y) 
# ----------------------------------------------------------------------
def axial_to_pixel(q, r, R): # (q,r) are axial coordinates which are used to represents the vertices of hexagon and R is used to represent the radius of hexagon
    """Flat-top hexagon axial coordinates -> Cartesian pixel coordinates."""
    x = 1.5 * R * q # calculate x coordinate of hexagon centre
    y = np.sqrt(3) * R * (r + q / 2.0) # y cooredinate of centre
    return x, y

# %%
# ----------------------------------------------------------------------
# STEP 2 - Cluster definition and the general co-channel label
# ----------------------------------------------------------------------
def build_cluster(i, j):
    """Cluster size N and the two lattice vectors that generate it."""
    N = i**2 + i * j + j**2
    A = (i, j)           # move i cells, move j cells
    B = (-j, i + j)      # A rotated by 60 degrees
    return N, A, B 

# %%
def make_labeller(i, j, N):
    """
    Return a function label(q, r) -> integer 0..N-1 that is identical on
    two cells if and only if they are co-channel. Works for ALL valid
    (i, j), including non-coprime pairs like (2, 2) for N = 12.
    """
    pair_to_id = {}

    def label(q, r):
        u = ((i + j) * q + j * r) % N
        v = (-j * q + i * r) % N
        key = (u, v)
        if key not in pair_to_id:
            pair_to_id[key] = len(pair_to_id)
        return pair_to_id[key]

    return label

# %%
# ----------------------------------------------------------------------
# STEP 3 + 4 + 5 - Generate/draw the reuse pattern, highlight co-channel
#                  cells, measure D, verify D/R = sqrt(3N)
# ----------------------------------------------------------------------
def generate_reuse_pattern(i, j, R=1.0, grid_span=7, save=True):
    """Build one complete reuse pattern and return (figure, results dict)."""
    N, A, B = build_cluster(i, j)
    label = make_labeller(i, j, N)
    # --- generate all cells inside a square viewing window ---
    centers = []
    window = grid_span * 1.55 * R
    for q in range(-grid_span - 2, grid_span + 3):
        for r in range(-grid_span - 2, grid_span + 3):
            x, y = axial_to_pixel(q, r, R)
            if abs(x) > window or abs(y) > window:
                continue
            centers.append((x, y, label(q, r)))
    n_labels = len({lab for _, _, lab in centers})
    assert n_labels == N, f"labelling produced {n_labels} labels, expected {N}"
    # --- draw the reuse pattern ---
    fig, ax = plt.subplots(figsize=(7.2, 7.2))
    for x, y, lab in centers:
        color = "#e74c3c" if lab == 0 else "#dfe6e9"
        ax.add_patch(RegularPolygon(
            (x, y), numVertices=6, radius=R * 0.985, orientation=0,
            edgecolor="black", facecolor=color, linewidth=0.8))
        letter = LETTERS[lab] if lab < 26 else str(lab)
        ax.text(x, y, letter, ha="center", va="center", fontsize=9,
                fontweight="bold" if lab == 0 else "normal",
                color="white" if lab == 0 else "black")

    # --- collect co-channel ("A") centres and measure D from geometry ---
    co = np.array([(x, y) for x, y, lab in centers if lab == 0])
    nearest = []
    for x0, y0 in co:
        d = np.hypot(co[:, 0] - x0, co[:, 1] - y0)
        d = d[d > 1e-9]
        if len(d):
            nearest.append(d.min())
    D_measured = float(np.median(nearest))
    D_theory = R * np.sqrt(3 * N)
    assert abs(D_measured - D_theory) < 1e-6 * R, "geometry check failed!"

    # --- draw one measured-D line, as Step 4 of the procedure asks ---
    p0 = co[np.argmin(np.hypot(co[:, 0], co[:, 1]))]          # nearest to centre
    d0 = np.hypot(co[:, 0] - p0[0], co[:, 1] - p0[1])
    d0[d0 < 1e-9] = np.inf
    p1 = co[np.argmin(d0)]
    ax.plot([p0[0], p1[0]], [p0[1], p1[1]], "b--", lw=2)
    ax.annotate(f"D = {D_measured:.3f}", ((p0[0]+p1[0])/2, (p0[1]+p1[1])/2),
                textcoords="offset points", xytext=(0, 10),
                color="blue", fontsize=11, fontweight="bold", ha="center")

    ax.set_xlim(-window - R, window + R)
    ax.set_ylim(-window - R, window + R)
    ax.set_aspect("equal"); ax.axis("off")
    ax.set_title(f"Frequency Reuse Pattern, N = {N}  (i = {i}, j = {j})   "
                 f"D/R = {D_measured/R:.3f}", fontsize=12)
    fname = f"reuse_N{N}.png"
    if save:
        fig.savefig(fname, dpi=200, bbox_inches="tight")

    # --- engineering numbers: first-tier SIR and per-cell capacity ---
    n_path = 4.0                        # urban path-loss exponent
    Q = D_theory / R                    # co-channel reuse ratio, sqrt(3N)
    sir_db = 10 * np.log10((Q ** n_path) / 6.0)   # 6 first-tier interferers
    total_channels = 420                # example licensed spectrum
    per_cell = total_channels // N

    return fig, dict(N=N, i=i, j=j, ratio_theory=D_theory / R,
                     ratio_measured=D_measured / R, sir_db=sir_db,
                     channels_per_cell=per_cell, file=fname)

def freq_to_channel(mhz): # to identify which frequency band belongs to which channel
    """Convert a centre frequency in MHz to a Wi-Fi channel number."""
    if 2412 <= mhz <= 2472:
        return (mhz - 2407) // 5          # 2.4 GHz: ch 1..13
    if mhz == 2484:
        return 14                          # Channel 14 is a special 2.4 GHz Wi-Fi channel associated with Japan and is not part of the normal 1–13 channel set used in most regions.
    if 5000 <= mhz <= 5900:
        return (mhz - 5000) // 5           # 5 GHz: ch 36, 40, ...
    return None
# %%


def scan_windows(): # to scan nearby networks and extracts SSID, Signal, Channel
    """Parse 'netsh wlan show networks mode=bssid' (Windows)."""
    out = subprocess.run(
        ["netsh", "wlan", "show", "networks", "mode=bssid"],
        capture_output=True, text=True).stdout
    nets, ssid = [], None
    for line in out.splitlines():
        line = line.strip()
        m = re.match(r"SSID \d+ : (.*)", line)
        if m:
            ssid = m.group(1).strip() or "<hidden>"
        m = re.match(r"Signal\s*:\s*(\d+)%", line)
        if m and ssid is not None:
            pct = int(m.group(1))
            rssi = pct / 2 - 100          # standard %-to-dBm approximation
            nets.append({"ssid": ssid, "rssi": rssi, "channel": None})
        m = re.match(r"Channel\s*:\s*(\d+)", line)
        if m and nets and nets[-1]["channel"] is None:
            nets[-1]["channel"] = int(m.group(1))
    return [n for n in nets if n["channel"] is not None]

def congestion(nets):
    """
    Count how many networks OVERLAP each 2.4 GHz channel, not just how many
    are centred on it: a network on channel c interferes with channels
    c-2 .. c+2. This is the number that actually predicts contention.
    """
    counts = {c: 0 for c in range(1, 14)}
    for n in nets:
        if n["channel"] <= 14:
            for c in range(max(1, n["channel"] - 2),
                           min(13, n["channel"] + 2) + 1):
                counts[c] += 1
    return counts

def plot_congestion(nets, fname="channel_congestion.png"):
    counts = congestion(nets)
    fig, ax = plt.subplots(figsize=(9, 4.5))
    colors = ["#2ecc71" if c in (1, 6, 11) else "#95a5a6"
              for c in counts]
    ax.bar(list(counts.keys()), list(counts.values()), color=colors)
    ax.set_xticks(range(1, 14))
    ax.set_xlabel("2.4 GHz channel number")
    ax.set_ylabel("Networks overlapping this channel")
    ax.set_title("Channel congestion (green = the non-overlapping "
                 "channels 1, 6, 11)")
    ax.grid(axis="y", alpha=0.3)
    fig.tight_layout()
    fig.savefig(fname, dpi=150)

    counts = congestion(nets)
    best = min((1, 6, 11), key=lambda c: counts[c])
    print("\nCongestion on the three non-overlapping channels:")

        print(f"   channel {c:>2}: {counts[c]} overlapping network(s)")
    print(f"[>] Recommended channel for a new 2.4 GHz router here: {best}")
    return best

# ----------------------------------------------------------------------
# STEP 2 - Average the repeat readings (and keep their spread)
# ----------------------------------------------------------------------
def average_readings(all_readings):
    """Mean and spread (std dev) of the repeat readings at each distance."""
    means = np.array([np.mean(r) for r in all_readings]) #to get mean of all the rssi readings
    spreads = np.array([np.std(r) for r in all_readings])# to calculate how much the readings vary from each other only for a particular reading
    return means, spreads 

# ----------------------------------------------------------------------
# STEP 3 - Theory: FSPL and the free-space prediction
# ----------------------------------------------------------------------
def fspl_db(d, f_mhz=F_MHZ):
    """Free Space Path Loss in dB (d in metres, f in MHz)."""
    return 20 * np.log10(d) + 20 * np.log10(f_mhz) - 27.55 # calculates free space path loss for every distance

def free_space_prediction(d, rssi_at_d0):
    """
    Free-space RSSI curve ANCHORED at your own d0 = 1 m measurement:
        RSSI(d) = RSSI(d0) - [FSPL(d) - FSPL(d0)]
                = RSSI(d0) - 20*log10(d/D0)
    Anchoring removes the unknown router transmit power and antenna gains
    from the comparison - only the SHAPE of the decay is being tested.
    """
    return rssi_at_d0 - 20 * np.log10(d / D0)
# %%

# ----------------------------------------------------------------------
# STEP 4 - Fit the Log-Distance Path Loss Model: find YOUR n
# ----------------------------------------------------------------------
def fit_path_loss_exponent(dists, rssi_means):
    """
    Log-distance model: RSSI(d) = RSSI(d0) - 10*n*log10(d/d0).
    Rearranged, RSSI is LINEAR in x = 10*log10(d/d0) with slope -n,
    so a least-squares straight-line fit gives n directly.
    Returns (n, rssi_d0_fit, sigma) where sigma is the shadowing spread
    X_sigma - the std dev of the residuals around the fitted line.
    """
    x = 10 * np.log10(dists / D0) 
    slope, intercept = np.polyfit(x, rssi_means, 1) 
    n = -slope 
    residuals = rssi_means - (slope * x + intercept)
    sigma = float(np.std(residuals))
    return n, intercept, sigma

def classify_environment(n):
    if n < 2.2:  return "free-space-like (open, unobstructed)"
    if n < 3.0:  return "lightly obstructed indoor / open corridor"
    if n < 4.0:  return "typical indoor with walls and furniture"
    return "heavily obstructed indoor (many walls / floors)"

import hashlib
import secrets
# ========================================================
# A3 ALGORITHM - AUTHENTICATION
# ========================================================

def A3(Ki, RAND):

    # Combine Ki and RAND
    combined = Ki + RAND

    # Process the combined value using SHA-256
    hash_value = hashlib.sha256(combined.encode()).hexdigest()

    # 32 bits = 8 hexadecimal characters
    result = hash_value[:8]

    return result


# ========================================================
# A8 ALGORITHM - KEY GENERATION
# ========================================================

def A8(Ki, RAND):

    # Combine Ki and RAND
    combined = Ki + RAND

    print("Ki + RAND =", combined)

    # Process Ki and RAND
    hash_value = hashlib.sha256(
        ("A8" + combined).encode()
    ).hexdigest()

    # 64 bits = 16 hexadecimal characters
    Kc = hash_value[:16]

    print("64-bit KC =", Kc)

    return Kc


# ========================================================
# A5 ALGORITHM - BLOCK ENCRYPTION
# ========================================================

def A5_encrypt(message, Kc):

    encrypted_blocks = []

    # Convert Kc from hexadecimal to bytes
    key = bytes.fromhex(Kc)

    # Convert message into bytes
    message_bytes = message.encode()

    # Divide message into blocks of 8 bytes
    block_size = 8

    for i in range(0, len(message_bytes), block_size):

        block = message_bytes[i:i + block_size]

        print("\nPlaintext block =", block.decode())

        # Encrypt block using XOR
        encrypted_block = bytes(
            block[j] ^ key[j % len(key)]
            for j in range(len(block))
        )

        print("Encrypted block =", encrypted_block.hex())

        encrypted_blocks.append(encrypted_block)

    # Combine all encrypted blocks
    encrypted_data = b''.join(encrypted_blocks)

    return encrypted_data


def A5_decrypt(encrypted_data, Kc):

    decrypted_blocks = []

    # Convert Kc from hexadecimal to bytes
    key = bytes.fromhex(Kc)

    # Divide encrypted data into blocks of 8 bytes
    block_size = 8

    for i in range(0, len(encrypted_data), block_size):

        block = encrypted_data[i:i + block_size]

        print("\nEncrypted block =", block.hex())

        # XOR again to decrypt
        decrypted_block = bytes(
            block[j] ^ key[j % len(key)]
            for j in range(len(block))
        )

        print("Decrypted block =", decrypted_block.decode())

        decrypted_blocks.append(decrypted_block)

    # Combine all decrypted blocks
    decrypted_data = b''.join(decrypted_blocks)

    return decrypted_data.decode()

# ---------------------------------------------------------
# Step 2: Metrics Quantifying Multipath
# ---------------------------------------------------------
def roughness(r):
    # Mean absolute step-to-step change: smooth traces -> small, multipath traces -> large
    valid = r[~np.isnan(r)]
    return float(np.mean(np.abs(np.diff(valid))))

def fade_depth(r):
    # Peak-to-null swing, dB
    valid = r[~np.isnan(r)]
    return float(valid.max() - valid.min())

def reversals(r):
    # % of steps where RSSI got STRONGER while moving further away
    valid = r[~np.isnan(r)]
    d = np.diff(valid)
    return 100.0 * float(np.sum(d > 0)) / d.size

# ---------------------------------------------------------
# Step 3: Two-Ray Theoretical Pattern
# ---------------------------------------------------------
def two_ray_pattern(pos_cm, start_extra_cm=0.0):
    lam_cm = WAVELENGTH_M * 100.0  # 12.5 cm at 2.4 GHz
    extra = start_extra_cm + 2.0 * pos_cm  # extra path grows as you move
    phase = 2 * np.pi * extra / lam_cm  # phase difference
    amp = np.abs(1.0 + np.exp(1j * phase))  # 0 (null) .. 2 (peak)
    return 20.0 * np.log10(np.clip(amp / 2.0, 1e-3, 1.0))  # dB relative to the peak

# ---------------------------------------------------------
# Step 4: Text Report
# ---------------------------------------------------------
def print_report(pos, indoor, outdoor, time_s, stat_rssi):
    r_ind = roughness(indoor)
    r_out = roughness(outdoor)
    f_ind = fade_depth(indoor)
    f_out = fade_depth(outdoor)
    rev_ind = reversals(indoor)
    rev_out = reversals(outdoor)

    print("=== Multipath fading survey ===")
    print("Wavelength at 2.4 GHz = 12.5 cm -> peaks/nulls repeat about every 6.2cm\n")
    print(f"{'Trace':<10} | {'Mean (dBm)':<11} | {'Spread(dB)':<11} | {'Roughness (dB)':<15} | {'FadeDepth(dB)':<14} | {'Reversals'}")
    print("-" * 75)
    print(f"{'Indoor':<10} | {np.mean(indoor):<11.1f} | {np.std(indoor):<11.1f} | {r_ind:<15.1f} | {f_ind:<14.1f} | {rev_ind:.0f}%")
    print(f"{'Outdoor':<10} | {np.nanmean(outdoor):<11.1f} | {np.nanstd(outdoor):<11.1f} | {r_out:<15.1f} | {f_out:<14.1f} | {rev_out:.0f}%")
    
    stat_mean = np.mean(stat_rssi)
    stat_spread = stat_rssi.max() - stat_rssi.min()
    print(f"\nStanding still: mean {stat_mean:.1f} dBm, but the reading still varied by {stat_spread:.0f} dB")
    print("without moving at all (people/doors changing the reflections).\n")
    
    ratio = r_ind / r_out if r_out > 0 else 0
    print("HEADLINE: the indoor trace is {:.1f}x rougher than outdoors ({:.1f} vs {:.1f} dB per step),".format(ratio, r_ind, r_out))
    print(f"with a {f_ind:.0f} dB peak-to-null swing over just {int(pos[-1]-pos[0])} cm, and RSSI got STRONGER on")
    print(f"{rev_ind:.0f}% of steps taken AWAY from the transmitter -- impossible under distance")
    print("loss alone. That is multipath fading.\n")

exp 10: Analysis of handover behaviour while moving (field observation)

import csv
import os
from collections import OrderedDict
import numpy as np
import matplotlib
import matplotlib.pyplot as plt

# A typical network setting. A handover is normally only allowed when the
# neighbour beats the serving cell by this margin (and keeps beating it for
# a short "time to trigger"). Both exist purely to stop ping-ponging.
HYSTERESIS_DB = 3.0
PING_PONG_WINDOW_S = 20.0  # returning to a cell within this = ping-pong

SAMPLE_CSV = """Time_s,ServingCellID,ServingRSRP_dBm,NeighborCellID,NeighborRSRP_dBm
0,12345,-72,67890,-101
5,12345,-76,67890,-97
10,12345,-81,67890,-92
15,12345,-85,67890,-88
20,12345,-89,67890,-84
25,67890,-83,12345,-91
30,67890,-86,12345,-88
35,12345,-87,67890,-89
40,67890,-85,12345,-90
45,67890,-80,12345,-95
50,67890,-78,54321,-99
55,67890,-83,54321,-90
60,67890,-88,54321,-85
65,54321,-82,67890,-92
70,54321,-77,67890,-97
"""

# ==============================================================================
# STEP 1 Data input
# ==============================================================================
def load_log(fname="handover_log.csv"):
    """Read the drive/walk log; create a sample file on the first run."""
    if not os.path.exists(fname):
        with open(fname, "w") as f:
            f.write(SAMPLE_CSV)
        print(f"[i] No {fname} found — a SAMPLE file was created representing")
        print("    a short walked route. Replace it with your own exported log,")
        print("    then rerun.")

    rows = []
    with open(fname) as f:
        for r in csv.reader(f):
            if not r or r[0].strip().startswith("Time"):
                continue  # skip header / blank lines
            rows.append({
                "t": float(r[0]),
                "cell": r[1].strip(),
                "rsrp": float(r[2]),
                "ncell": r[3].strip() if len(r) > 3 and r[3].strip() else "",
                "nrsrp": float(r[4]) if len(r) > 4 and r[4].strip() else np.nan,
            })
    return sorted(rows, key=lambda r: r["t"])


# ==============================================================================
# STEP 2 Find the handovers
# ==============================================================================
def detect_handovers(rows):
    """
    A handover is simply a change of serving Cell ID between two consecutive
    samples. For each one we also record the TRIGGER MARGIN measured in the
    sample immediately before the switch:
        margin = (neighbour RSRP) - (serving RSRP)
    That is how much better the target cell looked at the moment the network
    decided to move you. A healthy handover shows a clearly positive margin.
    """
    events = []
    for prev, cur in zip(rows, rows[1:]):
        if prev["cell"] == cur["cell"]:
            continue  # still on the same tower

        # was the cell we moved to the neighbour being reported before?
        margin = np.nan
        if prev["ncell"] == cur["cell"] and not np.isnan(prev["nrsrp"]):
            margin = prev["nrsrp"] - prev["rsrp"]

        events.append({
            "t": cur["t"],
            "from": prev["cell"],
            "to": cur["cell"],
            "rsrp_before": prev["rsrp"],
            "rsrp_after": cur["rsrp"],
            "margin": margin,
        })
    return events


def classify_handover(margin):
    """Judge a handover by the margin that triggered it."""
    if np.isnan(margin):
        return "unknown (no neighbour data)"
    if margin >= HYSTERESIS_DB:
        return "healthy"
    if margin > 0:
        return "weak margin"
    return "unjustified (target no better)"


# ==============================================================================
# STEP 3 Ping-pong detection
# ==============================================================================
def detect_ping_pong(events, window=PING_PONG_WINDOW_S):
    """
    A ping-pong is handing back to a cell you have just left, within a short
    window: A -> B then B -> A soon after. It wastes signalling, briefly
    interrupts the data flow, and is exactly what hysteresis and
    time-to-trigger are designed to prevent.
    """
    pings = []
    for a, b in zip(events, events[1:]):
        if b["to"] == a["from"] and (b["t"] - a["t"]) <= window:
            pings.append({
                "t1": a["t"],
                "t2": b["t"],
                "cellA": a["from"],
                "cellB": a["to"],
                "gap": b["t"] - a["t"]
            })
    return pings


def hysteresis_sweep(events, margins=np.arange(0, 9, 1.0)):
    """
    'What if the network had used a bigger margin?'
    For each candidate hysteresis value, count how many of the observed
    handovers would NOT have been allowed (because their trigger margin was
    too small). This shows how a larger margin suppresses marginal switches —
    the ones that cause ping-ponging — at the cost of holding on to a weak
    cell longer.
    """
    known = [e for e in events if not np.isnan(e["margin"])]
    out = []
    for m in margins:
        blocked = sum(1 for e in known if e["margin"] < m)
        out.append((float(m), blocked, len(known)))
    return out


# ==============================================================================
# STEP 4 Stability: dwell time and handover rate
# ==============================================================================
def dwell_times(rows):
    """How long each cell served you, in seconds (summed over visits)."""
    dwell = OrderedDict()
    for prev, cur in zip(rows, rows[1:]):
        dwell[prev["cell"]] = dwell.get(prev["cell"], 0.0) + (cur["t"] - prev["t"])
    return dwell


def stability(rows, events):
    duration = rows[-1]["t"] - rows[0]["t"] if len(rows) > 1 else 0.0
    rate = (len(events) / duration * 60.0) if duration else 0.0
    return duration, rate


# ==============================================================================
# STEP 5 Report
# ==============================================================================
def print_report(rows, events, pings):
    duration, rate = stability(rows, events)
    print("\n=== Handover analysis ===")
    print(f" Route logged for {duration:.0f} s, {len(rows)} samples, "
          f"{len(set(r['cell'] for r in rows))} distinct cells visited.")
    print(f" {len(events)} handover(s) -> {rate:.1f} per minute.")

    if events:
        hdr = " Time(s) From    -> To      RSRP before/after Margin Verdict"
        print("\n" + hdr)
        print("-" * len(hdr))
        for e in events:
            margin_str = "  n/a" if np.isnan(e["margin"]) else f"{e['margin']:+5.0f}"
            print(f" {e['t']:>6.0f}  {e['from']:<7} -> {e['to']:<7} "
                  f"{e['rsrp_before']:>6.0f} / {e['rsrp_after']:<6.0f} "
                  f"{margin_str} dB  {classify_handover(e['margin'])}")

    print("\n Dwell time per cell:")
    for cell, secs in dwell_times(rows).items():
        print(f"  cell {cell:<8} {secs:>5.0f} s")

    if pings:
        print(f"\n PING-PONG detected ({len(pings)} episode(s)):")
        for p in pings:
            print(f"  {p['cellA']} -> {p['cellB']} at t={p['t1']:.0f} s, "
                  f"back to {p['cellA']} at t={p['t2']:.0f} s (only {p['gap']:.0f} s later)")
        print("  The phone bounced between towers whose signals were nearly equal.")
        print("  This is precisely what a hysteresis margin and time-to-trigger")
        print("  are designed to prevent.")
    else:
        print("\n No ping-ponging detected — handovers were stable.")

    bad = [e for e in events if not np.isnan(e["margin"]) and e["margin"] < HYSTERESIS_DB]
    if bad:
        print(f"\nHEADLINE: {len(bad)} of {len(events)} handovers fired with a margin below the")
        print(f"  {HYSTERESIS_DB:.0f} dB hysteresis threshold — the target cell was barely better, or")
        print("  not better at all. Those marginal switches are what produce ping-ponging,")
        print("  wasted signalling and the brief data stalls you may have felt on the route.")

def plot_rsrp_timeline(rows, events, fname="handover_rsrp_timeline.png"):
    """The centerpiece: serving vs neighbour RSRP with handovers marked."""
    t = [r["t"] for r in rows]
    serving = [r["rsrp"] for r in rows]
    neighbour = [r["nrsrp"] for r in rows]

    fig, ax = plt.subplots(figsize=(9.0, 5.4))
    ax.plot(t, serving, "o-", color="tab:blue", lw=2.2, markersize=6, label="Serving cell RSRP")

    if not all(np.isnan(neighbour)):
        ax.plot(t, neighbour, "s--", color="tab:orange", lw=1.8, markersize=5, alpha=0.85, label="Best neighbour RSRP")

    for e in events:
        ax.axvline(e["t"], color="tab:red", ls=":", lw=1.5, alpha=0.8)
        ax.annotate(f"{e['from']}->{e['to']}", (e["t"], max(serving) + 1),
                    rotation=90, fontsize=7.5, color="firebrick", ha="right", va="top")

    ax.set_xlabel("Time along the route (s)")
    ax.set_ylabel("RSRP (dBm)")
    ax.set_title("Handover events: the neighbour overtakes the serving cell")
    ax.grid(alpha=0.3)
    ax.legend(loc="lower left", fontsize=9)
    fig.tight_layout()
    fig.savefig(fname, dpi=150)
    return fig


def plot_cell_timeline(rows, fname="handover_cell_timeline.png"):
    """Which tower served you at each moment."""
    cells = list(OrderedDict.fromkeys(r["cell"] for r in rows))
    idx = {c: i for i, c in enumerate(cells)}

    t = [r["t"] for r in rows]
    y = [idx[r["cell"]] for r in rows]

    fig, ax = plt.subplots(figsize=(9.0, 4.4))
    ax.step(t, y, where="post", color="tab:purple", lw=2.5)
    ax.scatter(t, y, s=28, color="tab:purple", zorder=3)
    ax.set_yticks(range(len(cells)))
    ax.set_yticklabels([f"cell {c}" for c in cells])
    ax.set_ylim(-0.5, len(cells) - 0.5)
    ax.set_xlabel("Time along the route (s)")
    ax.set_title("Serving Cell ID over time (each step is a handover)")
    ax.grid(alpha=0.3, axis="x")
    fig.tight_layout()
    fig.savefig(fname, dpi=150)
    return fig


def plot_margins(events, fname="handover_margins.png"):
    """Trigger margin per handover, against the hysteresis threshold."""
    known = [e for e in events if not np.isnan(e["margin"])]
    if not known:
        return None

    labels = [f"{e['from']}->{e['to']}\n(t={e['t']:.0f}s)" for e in known]
    margins = [e["margin"] for e in known]
    cols = ["tab:green" if m >= HYSTERESIS_DB else "tab:orange" if m > 0 else "tab:red" for m in margins]

    x = np.arange(len(known))
    fig, ax = plt.subplots(figsize=(8.8, 5.2))
    bars = ax.bar(x, margins, color=cols, edgecolor="black", lw=0.5)
    ax.axhline(HYSTERESIS_DB, ls="--", color="tab:blue", label=f"hysteresis threshold {HYSTERESIS_DB:.0f} dB")
    ax.axhline(0, color="black", lw=1)

    for b, m in zip(bars, margins):
        ax.text(b.get_x() + b.get_width() / 2,
                m + (0.15 if m >= 0 else -0.45),
                f"{m:+.0f}", ha="center",
                va="bottom" if m >= 0 else "top",
                fontsize=9, fontweight="bold")

    ax.set_xticks(x)
    ax.set_xticklabels(labels, fontsize=8)
    ax.set_ylabel("Trigger margin (dB)\nneighbour - serving")
    ax.set_title("Was each handover justified? Margin at the moment of switching")
    ax.legend()
    ax.grid(axis="y", alpha=0.3)
    fig.tight_layout()
    fig.savefig(fname, dpi=150)
    return fig


def plot_hysteresis_sweep(events, fname="hysteresis_sweep.png"):
    """How many marginal handovers a larger hysteresis would have blocked."""
    sweep = hysteresis_sweep(events)
    if not sweep or sweep[0][2] == 0:
        return None

    m = [s[0] for s in sweep]
    blocked = [s[1] for s in sweep]
    total = sweep[0][2]

    fig, ax = plt.subplots(figsize=(8.8, 5.0))
    ax.plot(m, blocked, "o-", color="tab:blue", lw=2.2, markersize=7)
    ax.axvline(HYSTERESIS_DB, ls="--", color="tab:red", label=f"typical setting {HYSTERESIS_DB:.0f} dB")
    ax.set_xlabel("Hysteresis margin required before handover (dB)")
    ax.set_ylabel(f"Handovers suppressed (of {total})")
    ax.set_title("Bigger hysteresis suppresses marginal handovers - and ping-ponging")
    ax.set_yticks(range(0, total + 1))
    ax.grid(alpha=0.3)
    ax.legend()
    fig.tight_layout()
    fig.savefig(fname, dpi=150)
    return fig
